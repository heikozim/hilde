# bash completion for pack
_pack() {
    local path_="pack" i w
    for (( i = 1; i < COMP_CWORD; i++ )); do
        w=${COMP_WORDS[i]}
        case "$path_:$w" in
            "pack:create"|"pack:c") path_="pack create" ;;
            "pack:list") path_="pack list" ;;
        esac
    done
    local prev=${COMP_WORDS[COMP_CWORD-1]} cur=${COMP_WORDS[COMP_CWORD]}
    case "$path_:$prev" in
        "pack create:--level"|"pack create:-l")
            COMPREPLY=( $(compgen -W "fast best" -- "$cur") )
            return ;;
        "pack create:--output"|"pack create:-o")
            COMPREPLY=( $(compgen -f -- "$cur") )
            return ;;
        "pack list:--key"|"pack list:-k")
            COMPREPLY=()
            return ;;
    esac
    case "$path_" in
        "pack")
            COMPREPLY=( $(compgen -W "--quiet -q --help --version create c list help" -- "$cur") ) ;;
        "pack create")
            COMPREPLY=( $(compgen -W "--quiet -q --level -l --output -o --help --version" -- "$cur") ) ;;
        "pack list")
            COMPREPLY=( $(compgen -W "--quiet -q --key -k --help --version" -- "$cur") ) ;;
    esac
}
complete -F _pack pack

#compdef pack
_pack() {
    local path_="pack" i w
    for (( i = 2; i < CURRENT; i++ )); do
        w=${words[i]}
        case "$path_:$w" in
            ("pack:create"|"pack:c") path_="pack create" ;;
            ("pack:list") path_="pack list" ;;
        esac
    done
    local prev=${words[CURRENT-1]}
    case "$path_:$prev" in
        ("pack create:--level"|"pack create:-l")
            compadd -- fast best
            return ;;
        ("pack create:--output"|"pack create:-o")
            _files
            return ;;
        ("pack list:--key"|"pack list:-k")
            _message value
            return ;;
    esac
    case "$path_" in
        ("pack")
            compadd -- --quiet -q --help --version create c list help ;;
        ("pack create")
            compadd -- --quiet -q --level -l --output -o --help --version ;;
        ("pack list")
            compadd -- --quiet -q --key -k --help --version ;;
    esac
}

# fish completion for pack
function _pack_path
    set -l path_ 'pack'
    for w in (commandline -opc)[2..-1]
        switch "$path_:$w"
            case "pack:create" "pack:c"
                set path_ 'pack create'
            case "pack:list"
                set path_ 'pack list'
        end
    end
    echo $path_
end
